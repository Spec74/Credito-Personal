# Strangler fig — staging/local.
# /api/v1/*, /health, /swagger → Credito.Modern.Api
# Resto → MVC .NET 4.8 (puerto del servidor local de VS / IIS Express).

upstream credito_modern_api {
    server credito-modern-api:8080;
}

# Puerto MVC: debe coincidir con donde corre Web en Visual Studio.
# - Servidor web de desarrollo (F5 típico): Web.csproj → DevelopmentServerPort (4779 en este repo).
# - IIS Express (perfil IIS): Web.csproj → IISUrl (52099 en este repo).
# Cambiar puerto: .\deploy\scripts\set-mvc-upstream-port.ps1 -Port TU_PUERTO
upstream credito_mvc_legacy {
    server host.docker.internal:4779;
}

server {
    listen 80;
    server_name _;
    client_max_body_size 16m;

    location /health {
        proxy_pass http://credito_modern_api;
        include /etc/nginx/proxy_params_strangler.conf;
    }

    location /swagger {
        proxy_pass http://credito_modern_api;
        include /etc/nginx/proxy_params_strangler.conf;
    }

    location /api/v1/ {
        proxy_pass http://credito_modern_api;
        include /etc/nginx/proxy_params_strangler.conf;
        proxy_connect_timeout 60s;
        proxy_send_timeout 120s;
        proxy_read_timeout 120s;
    }

    # MVC legado: headers inline (un solo archivo montado → menos fallos al añadir volúmenes).
    location / {
        proxy_pass http://credito_mvc_legacy;
        proxy_http_version 1.1;
        proxy_set_header Host localhost:4779;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_redirect http://localhost:4779/ http://$http_host/;
        proxy_connect_timeout 120s;
        proxy_send_timeout 120s;
        proxy_read_timeout 120s;
    }
}
