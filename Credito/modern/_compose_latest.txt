# Strangler staging — desde carpeta modern/:
#   docker compose -f deploy/docker-compose.strangler.yml --env-file deploy/.env up --build
#
# Puertos:
#   9080 — nginx (fachada única: MVC + /api/v1)
#   5080 — API directa (depuración, sin proxy)
#
# Smoke: .\deploy\scripts\smoke-strangler-proxy.ps1

services:
  credito-modern-api:
    build:
      context: ..
      dockerfile: Dockerfile
    environment:
      ASPNETCORE_ENVIRONMENT: Staging
      CreditoDatabase__ConnectionString: ${CREDITO_DB_CONNECTION_STRING}
      Jwt__SigningKey: ${JWT_SIGNING_KEY}
      Jwt__Issuer: ${JWT_ISSUER:-Credito.Modern.Staging}
      Jwt__Audience: ${JWT_AUDIENCE:-Credito.Modern.Staging}
      Jwt__RefreshAudience: ${JWT_REFRESH_AUDIENCE:-Credito.Modern.Staging.Refresh}
      Hosting__AllowDevToken: "true"
      Hosting__DisableHttpsRedirection: "true"
    ports:
      - "5080:8080"
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8080/health"]
      interval: 15s
      timeout: 5s
      retries: 5
      start_period: 20s

  strangler-proxy:
    image: nginx:1.27-alpine
    depends_on:
      credito-modern-api:
        condition: service_healthy
    ports:
      - "9080:80"
    volumes:
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf:ro
    extra_hosts:
      - "host.docker.internal:host-gateway"
