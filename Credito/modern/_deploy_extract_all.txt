# El MVC .NET 4.8 debe estar accesible en host.docker.internal:80 (ajustar nginx/default.conf).

services:
  credito-modern-api:
    build:
      context: ..
      dockerfile: Dockerfile
    environment:
      ASPNETCORE_ENVIRONMENT: Development
      CreditoDatabase__ConnectionString: ${CREDITO_DB_CONNECTION_STRING:-}
    ports:
      - "5080:8080"

  strangler-proxy:
    image: nginx:1.27-alpine
    depends_on:
      - credito-modern-api
    ports:
      - "9080:80"
    volumes:
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf:ro
